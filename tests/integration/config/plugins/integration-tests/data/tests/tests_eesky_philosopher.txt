			sprite "ship/star barge"

				cost 190000
				mass 10

				drag 1
				"engine capacity" 400
				"fuel capacity" 300

				hull 1000

				"required crew" 1
				shields 600

				"weapon capacity" 200
				"thrust" 50

test "eesky Philosopher Plugin Loaded"
	status active
	description "The eesky-philosopher plugin is installed for eesky tests."
	sequence
		inject "eesky Philosopher Test Save"
		call "Load First Savegame"
		assert
			"installed plugin: eesky-philosopher" == 1



test "eesky Stranger 1 Records Answer"
	status active
	description "The first Stranger meeting records the chosen answer and nudges the matching trait."
	sequence
		inject "eesky Philosopher Test Save"
		call "Load First Savegame"
		apply
			"eesky: debug: force stranger" = 1
		call "Trigger Ruin Spaceport Mission"
		# "That depends on who decides what 'better' means."
		input
			key 3
		input
			key "Return"
		assert
			"eesky: answer: stranger 1" == 3
			"eesky: trait: curious" == 1
			"eesky: trait: utilitarian" == 0
			"eesky: Stranger 1: offered" == 1
			"eesky: debug: force stranger" == 0
			"stat: conversation choice" == 1



test "eesky Stranger 2 Reads Play Statistics"
	status active
	description "The second Stranger meeting branches on play statistics and records what the player wants."
	sequence
		inject "eesky Philosopher Test Save"
		call "Load First Savegame"
		apply
			"eesky: Stranger 1: offered" = 1
			"stat: destroyed" = 12
			"eesky: debug: force stranger" = 1
		call "Trigger Ruin Spaceport Mission"
		# "A good fight."
		input
			key 1
		input
			key "Return"
		assert
			"eesky: answer: stranger 2" == 1
			"eesky: wants: danger" == 1
			"eesky: trait: bold" == 1
			"eesky: Stranger 2: offered" == 1



test "eesky Local Records Answer"
	status active
	description "A one-off local encounter records its answer and does not repeat."
	sequence
		inject "eesky Philosopher Test Save"
		call "Load First Savegame"
		apply
			"eesky: debug: force ledger" = 1
		call "Trigger Ruin Spaceport Mission"
		# "Bury it. Three hundred families matter more than one crook."
		input
			key 2
		input
			key "Return"
		assert
			"eesky: answer: ledger" == 2
			"eesky: trait: utilitarian" == 1
			"eesky: local: ledger: offered" == 1



test "eesky Beggar Takes Payment"
	status active
	description "A wealthy player is asked for more, and giving costs real credits."
	sequence
		inject "eesky Philosopher Test Save"
		call "Load First Savegame"
		apply
			"eesky: debug: force beggar" = 1
			"test: credits before" = "credits"
		call "Trigger Ruin Spaceport Mission"
		# "(Give him 10,000 credits.)"
		input
			key 1
		input
			key "Return"
		assert
			"eesky: beggar: gave" == 10000
			"eesky: beggar: met" == 1
			"eesky: trait: generous" == 2
			"test: credits before" - "credits" == 10000



test "eesky Grey Ship Answers Hail"
	status active
	description "The grey ship waits in the system, and hailing it starts its conversation instead of the normal hail panel."
	sequence
		inject "eesky Philosopher Test Save"
		call "Load First Savegame"
		apply
			"eesky: debug: force grey ship" = 1
		# Landing offers (and silently accepts) the grey ship mission.
		call "Depart"
		navigate
			"travel destination" Ruin
		call "Land"
		assert
			"eesky: grey ship: seen" == 1
		# Take off, then target and hail ships until the grey ship answers.
		call "Depart"
		# Hailing is handled by the flight panel's key bindings, so press the
		# default "talk to selected ship" key after selecting the next ship.
		label "hailing"
		input
			command target
		input
			key t
		branch "hailing"
			not "eesky: grey ship: hailed"
		# "Of course. Wrong is wrong, watched or not."
		input
			key 1
		input
			key "Return"
		assert
			"eesky: answer: grey ship" == 1
			"eesky: trait: utilitarian" == -1
			"stat: hailed ship: Independent" == 1
		# The hail ends the mission; it is removed once the player lands.
		navigate
			"travel destination" Ruin
		call "Land"
		assert
			"eesky: The Grey Ship: failed" == 1
			"eesky: The Grey Ship: active" == 0
