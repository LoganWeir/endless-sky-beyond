# Give the hyperspace arrival flash some images. Headless tests don't draw
# frames, so this checks that "jump flash" data loads and that jumping with it
# works; the drawing itself has to be checked by eye.
"jump flash"
	sprite "ship/star barge"
	sprite "ship/shuttle"



test "eesky Jump Flash Data Loads"
	status active
	description "Jumping with plugin-provided flash images works, and the jump is recorded."
	sequence
		inject "Three Earthly Barges Save"
		call "Load First Savegame"
		call "Depart"
		navigate
			travel "Alpha Centauri"
		input
			command jump
		label "jumping"
		branch "jumping"
			not "flagship system: Alpha Centauri"
		assert
			"flagship system: Alpha Centauri" == 1
			"stat: jumped: Alpha Centauri" == 1
			"stat: jump method: hyperdrive" == 1
